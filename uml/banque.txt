=====================================================================
1. CLASSES ET ATTRIBUTSj
=====================================================================

adresse_type <<type objet>
  - num_rue : NUMBER
  - rue     : VARCHAR2(80)
  - ville   : VARCHAR2(40)
  - cp      : VARCHAR2(10)

tel_vry_type <<VARRAY(3) OF VARCHAR2(20)>>
  [1] domicile
  [2] portable 1
  [3] portable 2
  (elements pouvant etre NULL ; tableau partiellement rempli)

conseiller_type <<type objet>>
  - num_conseiller : NUMBER
  - nom            : VARCHAR2(30)
  - prenom         : VARCHAR2(30)
  - specialite     : VARCHAR2(40)
  - date_embauche  : DATE
  - salaire        : NUMBER(10,2)

agence_type <<type objet>>
  - code            : VARCHAR2(6)
  - nom             : VARCHAR2(40)
  - ville           : VARCHAR2(40)
  - ref_responsable : REF conseiller_type

client_type <<type objet>>
  - num_client     : NUMBER
  - nom            : VARCHAR2(30)
  - prenom         : VARCHAR2(30)
  - date_naissance : DATE
  - num_secu       : VARCHAR2(20)
  - adresse        : adresse_type
  - telephones     : tel_vry_type
  - email          : VARCHAR2(60)
  - date_entree    : DATE
  - ref_conseiller : REF conseiller_type
  - ref_parrain    : REF client_type          (ajoute par ALTER TYPE, E10)

affectation <<table relationnelle>>
  - id_affectation : NUMBER
  - ref_conseiller : REF conseiller_type
  - ref_agence     : REF agence_type
  - date_debut     : DATE
  - date_fin       : DATE

compte_type <<type objet, NOT FINAL NOT INSTANTIABLE>>
  - num_compte     : VARCHAR2(10)
  - solde          : NUMBER(12,2)
  - date_ouverture : DATE
  - statut         : VARCHAR2(10)   ACTIF | BLOQUE | DORMANT
  - ref_client     : REF client_type
  - ref_agence     : REF agence_type

cptCourant_type <<type objet, UNDER compte_type>>
  - decouvert   : NUMBER(12,2)
  - nb_op_cb    : NUMBER
  - signataires : signataire_nt_type

cptEpargne_type <<type objet, UNDER compte_type>>
  - taux_interet  : NUMBER(5,2)     strictement < 3.5
  - plafond_depot : NUMBER(12,2)

cptTerme_type <<type objet, UNDER compte_type>>
  - montant_bloque : NUMBER(12,2)
  - duree_mois     : NUMBER          6 | 12 | 24 | 36
  - date_echeance  : DATE
  - taux_garanti   : NUMBER(5,2)

signataire_elt_nt_type <<element de nested table>>
  - ref_client : REF client_type
  - droit      : CHAR(1)             R | D | X

operation_type <<type objet>>
  - id_op      : NUMBER
  - ref_client : REF client_type
  - ref_compte : REF cptCourant_type
  - date_op    : DATE
  - montant    : NUMBER(12,2)        > 0 credit, < 0 debit
  - type_op    : VARCHAR2(10)        DEPOT | RETRAIT | VIREMENT | CB
  - commission : NUMBER(12,2)

type_stat_elt_nt_type <<element de nested table, E11>>
  - type_op       : VARCHAR2(10)
  - nb            : NUMBER
  - montant_total : NUMBER(14,2)

mois_elt_vry_type <<element de VARRAY, E11>>
  - mois          : NUMBER
  - nb_operations : NUMBER
  - types         : type_stat_nt_type

releves <<table relationnelle, E11>>
  - annee    : NUMBER
  - mois_vry : mois_vry_type         VARRAY(12)

agence_vor_type <<type objet de vue, E12>>
  - code        : VARCHAR2(6)
  - nom         : VARCHAR2(40)
  - conseillers : conseiller_nt_type
  - comptes     : cptCourant_nt_type

=====================================================================
3. ASSOCIATIONS ET CARDINALITES
=====================================================================

client_type 1 -------- 0..1 client_type          association recursive
                                                 "parrain / filleul"

client_type 1 -------- 0..1 conseiller_type      "conseiller attitre"

agence_type 1 -------- 1    conseiller_type      "responsable"

conseiller_type 1 ---- 0..* affectation          "historique des postes"
agence_type    1 ---- 0..* affectation

client_type 1 -------- 0..* compte_type          "titulaire (1 seul)"
agence_type 1 -------- 0..* compte_type          "agence de domiciliation"

compte_type 1 -------- 0..* operation_type       "mouvements"
client_type 1 -------- 0..* operation_type       "auteur de l'operation"

cptCourant_type 1 ---- 0..* signataire_elt_nt_type
                                                 classe d'association
                                                 avec attribut "droit"
client_type 1 -------- 0..* signataire_elt_nt_type

client_type 1 -------- 0..3 tel_vry_type         aggregation (VARRAY)
client_type 1 -------- 1    adresse_type         composition

releves 1 ------------ 0..12 mois_elt_vry_type
mois_elt_vry_type 1 -- 0..*  type_stat_elt_nt_type

agence_vor_type 1 ---- 0..* conseiller_type      (vue objet)
agence_vor_type 1 ---- 0..* cptCourant_type      (vue objet)

=====================================================================
4. CORRESPONDANCE AVEC LES EXIGENCES DE A1
=====================================================================

[OK] Hierarchie d'heritage des comptes
     compte_type (NOT FINAL NOT INSTANTIABLE)
       <- cptCourant_type
       <- cptEpargne_type
       <- cptTerme_type

[OK] Associations avec cardinalites
     client 1 - 0..* compte ; agence 1 - 0..* compte ;
     client 1 - 0..1 conseiller ; agence 1 - 1 conseiller ;
     conseiller 1 - 0..* affectation ; agence 1 - 0..* affectation.

[OK] Classe d'association des signataires (avec le droit)
     signataire_elt_nt_type (ref_client, droit)
     cptCourant_type 1 - 0..* signataire_elt_nt_type.

[OK] Association n-aire des operations
     operation_type relie client, compte courant et date
     (montant, type, commission).

[OK] Association recursive de parrainage
     client_type.ref_parrain : REF client_type
     (ajoute par ALTER TYPE ... CASCADE, question E10).

[OK] Aggregation des telephones
     client_type.telephones : tel_vry_type (VARRAY de 3).

[OK] Methodes
     Elles figurent dans les classes et sont ajoutees par
     ALTER TYPE ... ADD MEMBER ... CASCADE (partie D), y compris
     fraisTenue ajoutee en D8 : racine puis OVERRIDING dans les 3 sous-types.
